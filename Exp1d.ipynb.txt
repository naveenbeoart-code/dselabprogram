        "import numpy as np\n",
        "\n",
        "# Array from a list and from a tuple\n",
        "a = np.array([1, 2, 3, 4])\n",
        "b = np.array((5, 6, 7, 8))                                  # array from a tuple ( )\n",
        "print(\"From list :\", a)\n",
        "print(\"From tuple:\", b)\n",
        "\n",
        "# Arithmetic operations and transpose\n",
        "print(\"a + b =\", a + b)\n",
        "print(\"a - b =\", a - b)\n",
        "print(\"a * b =\", a * b)\n",
        "print(\"b / a =\", b / a)\n",
        "m = np.array([[1, 2, 3], [4, 5, 6]])\n",
        "print(\"Transpose:\")\n",
        "print(m.T)                                                  # .T = transpose (rows become columns)\n",
        "\n",
        "# Centigrade to Fahrenheit and back\n",
        "c = np.array([0, 25, 37, 100])\n",
        "f = c * 9 / 5 + 32                                          # F = C x 9/5 + 32 for every value\n",
        "print(\"Fahrenheit:\", f)\n",
        "print(\"Centigrade:\", (f - 32) * 5 / 9)\n",
        "\n",
        "# Real and imaginary parts of complex numbers\n",
        "z = np.array([1 + 2j, 3 - 4j])\n",
        "print(\"Real:\", z.real, \" Imaginary:\", z.imag)               # .real and .imag parts\n",
        "\n",
        "# 5-dimensional array\n",
        "d = np.array([1, 2, 3], ndmin=5)                            # ndmin = minimum dimensions\n",
        "print(\"Dimensions:\", d.ndim)\n",
        "\n",
        "# Sort a boolean array\n",
        "t = np.array([True, False, True, False])\n",
        "print(\"Sorted boolean:\", np.sort(t))                        # False comes before True\n",
        "\n",
        "# Save an array to a CSV file\n",
        "np.savetxt('data/numbers.csv', m, delimiter=',', fmt='%d')  # save array to CSV file\n",
        "print(\"Saved to data/numbers.csv\")"
